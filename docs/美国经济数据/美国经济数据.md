<script>
(() => {
  const BASE_W = 670;
  const BASE_H = 512;

  document.querySelectorAll('.fred-frame').forEach(frameEl => {

    const stage = frameEl.querySelector('.fred-stage');

    function apply(){
      const w = frameEl.getBoundingClientRect().width;
      const s = Math.min(1, w / BASE_W);

      stage.style.transform = `scale(${s})`;
      frameEl.style.height = `${BASE_H * s}px`;
    }

    apply();
    requestAnimationFrame(apply);

    new ResizeObserver(apply).observe(frameEl);
    window.addEventListener('orientationchange', () => setTimeout(apply, 50), { passive: true });
  });
})();
</script>
